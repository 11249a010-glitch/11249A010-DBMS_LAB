CREATE TABLE CARR (
    CARR_ID NUMBER(5) CONSTRAINT PK_CARR PRIMARY KEY,
    BRAND_NAME VARCHAR2(30) CONSTRAINT NN_BRAND_NAME NOT NULL,
    MODEL_NAME VARCHAR2(30) CONSTRAINT NN_MODEL_NAME NOT NULL,
    REG_NO VARCHAR2(15) CONSTRAINT UQ_REG_NO UNIQUE,
    PRICE NUMBER(10,2) CONSTRAINT CHK_PRICE CHECK (PRICE > 0),
    COLOR VARCHAR2(20) DEFAULT 'White'
);
