CREATE TABLE COURSE (
    COURSE_ID NUMBER(5) CONSTRAINT PK_COURSE PRIMARY KEY,
    COURSE_NAME VARCHAR2(50) CONSTRAINT NN_COURSE_NAME NOT NULL,
    DEPARTMENT VARCHAR2(50),
    DURATION VARCHAR2(20) DEFAULT '6 Months',
    CREDITS NUMBER(2) CONSTRAINT CHK_CREDITS CHECK (CREDITS BETWEEN 1 AND 6),
    FEES NUMBER(10,2) CONSTRAINT CHK_FEES CHECK (FEES > 0),
    COURSE_CODE VARCHAR2(10) CONSTRAINT UQ_COURSE_CODE UNIQUE
);
